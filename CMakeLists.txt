cmake_minimum_required(VERSION 3.20)
project(raftkv VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(ROCKSDB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_BENCHMARK_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_CORE_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_GFLAGS OFF CACHE BOOL "" FORCE)
set(WITH_SNAPPY OFF CACHE BOOL "" FORCE)
set(WITH_LZ4 OFF CACHE BOOL "" FORCE)
set(WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(WITH_ZSTD OFF CACHE BOOL "" FORCE)
set(WITH_BZ2 OFF CACHE BOOL "" FORCE)
set(FAIL_ON_WARNINGS OFF CACHE BOOL "" FORCE)
set(PORTABLE ON CACHE BOOL "" FORCE)
FetchContent_Declare(rocksdb GIT_REPOSITORY https://github.com/facebook/rocksdb.git GIT_TAG v9.7.3 GIT_SHALLOW TRUE)
# Use a pinned source tag for the header-only JSON dependency.
FetchContent_Declare(nlohmann GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(rocksdb nlohmann)
add_executable(raftkv src/main.cpp)
target_link_libraries(raftkv PRIVATE rocksdb nlohmann_json::nlohmann_json)
target_compile_options(raftkv PRIVATE -Wall -Wextra -Wpedantic)
option(SANITIZE "Enable address/undefined sanitizers on application code" OFF)
if(SANITIZE)
 target_compile_options(raftkv PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(raftkv PRIVATE -fsanitize=address,undefined)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME cluster COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/cluster.py $<TARGET_FILE:raftkv>)
set_tests_properties(cluster PROPERTIES TIMEOUT 120)
install(TARGETS raftkv RUNTIME DESTINATION bin)
add_test(NAME safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/safety.py $<TARGET_FILE:raftkv>)
set_tests_properties(safety PROPERTIES TIMEOUT 30)
